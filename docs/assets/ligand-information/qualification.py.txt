def selected(path: str) -> bool:
    return path.startswith(PREFIXES) or path in CONFIGS or path in TOOLS or path in DATA


def digest(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def browser_report_records(browser: dict[str, Any]) -> list[dict[str, Any]]:
    """Use the existing complete test identity, including project and parent titles."""
    records: list[dict[str, Any]] = []

    def walk(suite: dict[str, Any], parents: tuple[str, ...] = ()) -> None:
        for spec in suite.get("specs", []):
            for test in spec["tests"]:
                results = test["results"]
                assert results, "A preceding browser case has no observed result"
                title = " \u203a ".join((*parents, spec["title"]))
                records.append(
                    {
                        "identity": (
                            f"[{test['projectName']}] \u203a "
                            f"tests/e2e/{spec['file']} \u203a {title}"
                        ),
                        "status": results[-1]["status"],
                        "expected_status": test["expectedStatus"],
                        "retries": len(results) - 1,
                    }
                )
        for child in suite.get("suites", []):
            walk(child, parents + ((suite["title"],) if suite.get("specs") else ()))

    for suite in browser["suites"]:
        walk(suite)
    return records


def safe_relative(name: str) -> str:
    path = Path(name)
    assert name and not path.is_absolute() and "\\" not in name
    assert path.as_posix() == name and all(part not in {"", ".", ".."} for part in name.split("/"))
    return name


def owned_file(root: Path, name: str) -> Path:
    path = root / safe_relative(name)
    assert root.resolve() == root and path.resolve() == path and path.is_file()
    assert not path.is_symlink(), f"Aliased qualification input: {name}"
    return path


def identity(path: Path) -> dict[str, Any]:
    body = path.read_bytes()
    return {"bytes": len(body), "sha256": digest(body)}


def require_pin(pin: str | None) -> str:
    assert isinstance(pin, str) and re.fullmatch(r"[a-f0-9]{64}", pin), (
        "Actual v0.38.0 public baseline provenance has not been pinned"
    )
    return pin


def validate_preceding_browser_report(
    browser: dict[str, Any], summary: dict[str, Any], expected: dict[str, Any]
) -> dict[str, Any]:
    """Validate actual pinned counts and exact exclusions, never prospective totals."""
    assert not browser.get("errors"), "The preceding browser report contains errors"
    for field in (
        "complete_browser_inventory_executed",
        "existing_exclusion_identities_unchanged",
        "clean_start",
        "sources_unchanged",
        "source_inventory_unchanged",
        "all_lossless_log_hashes_valid",
    ):
        assert summary[field] is True, field
    assert summary["new_exclusions"] == summary["failures_flakes_retries"] == 0
    assert summary["complete_steps"] == len(REQUIRED_STEPS) == 43 and summary["python_skips"] == 0
    assert type(expected["source_count"]) is int and expected["source_count"] > 0
    assert summary["source_count"] == expected["source_count"]
    stats = browser["stats"]
    assert stats["unexpected"] == stats["flaky"] == 0
    records = browser_report_records(browser)
    identities = [record["identity"] for record in records]
    assert identities and len(identities) == len(set(identities)), "Duplicate preceding case"
    assert all(
        record["retries"] == 0
        and record["status"] in {"passed", "skipped"}
        and record["status"] == record["expected_status"]
        for record in records
    ), "Incomplete, retried or unexpected preceding case"
    excluded = sorted(record["identity"] for record in records if record["status"] == "skipped")
    passed = sum(record["status"] == "passed" for record in records)
    for field, actual in (
        ("browser_inventory_count", len(records)),
        ("browser_passed", passed),
        ("browser_exclusions", len(excluded)),
    ):
        assert type(expected[field]) is int and expected[field] >= 0
        assert actual == summary[field] == expected[field], field
    assert passed == stats["expected"] and len(excluded) == stats["skipped"]
    excluded_digest = digest(json.dumps(excluded).encode())
    assert excluded_digest == expected["excluded_identities_sha256"], (
        "Predecessor exclusion set changed"
    )
    return {
        "browser_inventory_count": len(records),
        "browser_passed": passed,
        "browser_exclusions": len(excluded),
        "excluded_identities": excluded,
        "excluded_identities_sha256": excluded_digest,
        "inventory_identities_sha256": digest(json.dumps(sorted(identities)).encode()),
    }


def validate_publication(provenance: dict[str, Any], preceding: dict[str, Any]) -> None:
    """Inspect the independently pinned publication receipt; no transport is repeated."""
    assert provenance["schema"] == "published-predecessor-browser-baseline-v1"
    expected = provenance["predecessor"]
    assert expected["tag"] == "v" + PREVIOUS_VERSION and expected["version"] == PREVIOUS_VERSION
    assert re.fullmatch(r"[a-f0-9]{40}", expected["commit"])
    assert all(preceding[k] == expected[k] for k in ("tag", "version", "commit"))
    public = provenance["public_asset"]
    url = urllib.parse.urlsplit(public["url"])
    prefix = f"/ManuelSe/Neistra/releases/download/v{PREVIOUS_VERSION}/"
    assert (
        url.scheme == "https" and url.netloc == "github.com" and not url.query and not url.fragment
    )
    assert url.path.startswith(prefix) and re.fullmatch(
        r"[a-zA-Z0-9._-]+\.zip", url.path[len(prefix) :]
    )
    assert type(public["bytes"]) is int and public["bytes"] > 0
    assert re.fullmatch(r"[a-f0-9]{64}", public["sha256"])
    assert provenance["nested_archive"]["all_crc_verified"] is True
    for record in (provenance["outer_manifest"], provenance["nested_archive"]):
        value = record.get("sha256", record.get("manifest_sha256"))
        assert re.fullmatch(r"[a-f0-9]{64}", value)
    outer = provenance["outer_manifest"]["member"]
    safe_relative(outer["path"])
    assert type(outer["bytes"]) is int and outer["bytes"] > 0
    assert re.fullmatch(r"[a-f0-9]{64}", outer["sha256"])
    verification = provenance["publication_verification"]
    assert re.fullmatch(r"[a-f0-9]{40}", verification["annotated_tag_object"])
    assert verification["reachable_on_remote_master"] is True
    assert verification["published"] is True and verification["draft"] is False
    assert verification["prerelease"] is False
    download = verification["download"]
    assert download["anonymous_request_without_authorization_header"] is True
    assert download["http_status"] == 200 and download["qualified_master"] == expected["commit"]
    assert download["bytes"] == public["bytes"] and download["sha256"] == public["sha256"]
    assert download["public_url"] == public["url"]
    assert download["all_zip_crcs_and_internal_manifest_hashes_valid"] is True


def preceding_browser_baseline(
    root: Path, preceding: dict[str, Any], *, provenance_pin: str | None = None
) -> dict[str, Any]:
    """Require original published v0.38.0 members and an independent provenance pin."""
    pin = require_pin(BASELINE_PROVENANCE_SHA256 if provenance_pin is None else provenance_pin)
    files = {relative: identity(owned_file(root, relative)) for relative in sorted(BASELINE_DATA)}
    provenance_path = BASELINE_DIRECTORY + "provenance.json"
    assert files[provenance_path]["sha256"] == pin, "Published predecessor provenance changed"
    provenance = json.loads((root / provenance_path).read_bytes())
    validate_publication(provenance, preceding)
    aliases = provenance["retained_members"]
    assert set(aliases) == set(BASELINE_ALIASES)
    assert len(set(aliases.values())) == 2, "Baseline members are aliased"
    members = provenance["nested_archive"]["members"]
    for alias, member in aliases.items():
        safe_relative(member)
        assert files[BASELINE_DIRECTORY + alias] == members[member], member
    summary_path = BASELINE_DIRECTORY + BASELINE_ALIASES[0]
    browser_path = BASELINE_DIRECTORY + BASELINE_ALIASES[1]
    summary = json.loads((root / summary_path).read_bytes())
    assert summary["commit"] == preceding["commit"]
    assert summary["application_version"] == PREVIOUS_VERSION
    browser = json.loads(gzip.decompress((root / browser_path).read_bytes()))
    expected = {
        **provenance["predecessor"],
        **{
            field: provenance[field]
            for field in (
                "browser_inventory_count",
                "browser_passed",
                "browser_exclusions",
                "excluded_identities_sha256",
            )
        },
    }
    return {
        "predecessor": provenance["predecessor"],
        "data_files": files,
        "provenance": provenance_path,
        "summary": summary_path,
        "browser": browser_path,
        "publication_verification": provenance["publication_verification"],
        **validate_preceding_browser_report(browser, summary, expected),
    }


def qualification_data_inventory(
    root: Path,
    *,
    provenance_pin: str | None = None,
    legacy_provenance_pin: str = LEGACY_PROVENANCE_SHA256,
) -> dict[str, dict[str, Any]]:
    """Freeze all six bodies independently of the new three-file browser authority."""
    pin = require_pin(BASELINE_PROVENANCE_SHA256 if provenance_pin is None else provenance_pin)
    files = {relative: identity(owned_file(root, relative)) for relative in sorted(DATA)}
    physical = [owned_file(root, name).stat() for name in sorted(DATA)]
    assert len({(value.st_dev, value.st_ino) for value in physical}) == len(DATA), (
        "Aliased DATA files"
    )
    assert files[BASELINE_DIRECTORY + "provenance.json"]["sha256"] == pin
    require_pin(legacy_provenance_pin)
    assert files[LEGACY_DIRECTORY + "provenance.json"]["sha256"] == legacy_provenance_pin
    legacy = json.loads((root / LEGACY_DIRECTORY / "provenance.json").read_bytes())
    assert legacy["predecessor"]["tag"] == "v0.37.0"
    assert legacy["predecessor"]["version"] == "0.37.0"
    members = legacy["nested_archive"]["members"]
    assert len(members) == 2
    assert {LEGACY_DIRECTORY + Path(member).name for member in members} == LEGACY_DATA - {
        LEGACY_DIRECTORY + "provenance.json"
    }
    for member, record in members.items():
        safe_relative(member)
        assert files[LEGACY_DIRECTORY + Path(member).name] == record
    return files


def verify_predecessor_catalogue(
    root: Path, preceding: dict[str, Any], receipt: Path, baseline: dict[str, Any]
) -> dict[str, str]:
    """Require annotated-tag, literal earlier catalogue, extraction and publication agreement."""
    assert preceding["tag"] == f"v{PREVIOUS_VERSION}" and preceding["version"] == PREVIOUS_VERSION
    tag = "refs/tags/" + preceding["tag"]
    assert (
        subprocess.check_output(["git", "cat-file", "-t", tag], cwd=root, text=True).strip()
        == "tag"
    )
    assert (
        subprocess.check_output(["git", "rev-parse", tag], cwd=root, text=True).strip()
        == (baseline["publication_verification"]["annotated_tag_object"])
    )
    commit = subprocess.check_output(["git", "rev-parse", tag + "^{}"], cwd=root, text=True).strip()
    assert commit == preceding["commit"] == baseline["predecessor"]["commit"]
    subprocess.run(
        ["git", "merge-base", "--is-ancestor", commit, "HEAD"],
        cwd=root,
        check=True,
        capture_output=True,
    )
    expected = tagged_identities(preceding["tag"], root)
    legacy = root / "docs/assets/ligand-profiles/qualification.py.txt"
    assert digest(legacy.read_bytes()) == expected[str(legacy.relative_to(root))]
    code = (
        "import json,runpy,sys; "
        "mapping=runpy.run_path(sys.argv[1])['tagged_identities'](sys.argv[2]); "
        "print(json.dumps(mapping,sort_keys=True))"
    )
    original = json.loads(
        subprocess.check_output(
            [sys.executable, "-B", "-c", code, str(legacy), preceding["tag"]], cwd=root, text=True
        )
    )
    assert expected == original, "Predecessor source catalogue was narrowed or expanded"
    assert receipt.resolve() == receipt and receipt.is_file() and not receipt.is_symlink()
    record = json.loads(receipt.read_bytes())
    assert record["tag"] == preceding["tag"] and record["commit"] == commit
    assert record["files"] == expected and len(expected) == baseline["predecessor"]["source_count"]
    assert extracted_identities(Path(preceding["root"]), expected) == expected
    return expected

def current_identities(root):
    names = subprocess.check_output(
        ["git", "ls-files", "--cached", "--others", "--exclude-standard"],
        cwd=root,
        text=True,
    ).splitlines()
    identities = {}
    for path in sorted(set(names)):
        if not selected(path):
            continue
        source = root / path
        assert not source.is_symlink(), f"Qualifying source is a symlink: {path}"
        if source.is_file():
            identities[path] = digest(source.read_bytes())
    validate_current_catalogue(identities)
    return identities


def tagged_identities(tag, root=None):
    root = Path.cwd() if root is None else root
    names = [
        p
        for p in subprocess.check_output(
            ["git", "ls-tree", "-r", "--name-only", tag],
            cwd=root,
            text=True,
        ).splitlines()
        if selected(p)
    ]
    objects = subprocess.check_output(
        ["git", "cat-file", "--batch"],
        cwd=root,
        input="".join(f"{tag}:{p}\n" for p in names).encode(),
    )
    identities = {}
    offset = 0
    for name in names:
        end = objects.index(b"\n", offset)
        _, kind, size = objects[offset:end].split()
        assert kind == b"blob", name
        start = end + 1
        offset = start + int(size)
        identities[name] = digest(objects[start:offset])
        assert objects[offset : offset + 1] == b"\n"
        offset += 1
    assert offset == len(objects)
    return identities


def extracted_identities(root, names):
    return {p: digest(owned_file(root, p).read_bytes()) for p in names}

def fixture_qualification_reports(recipe, root):
    """Require and bind every saved scientific case in the owned gate workspace."""
    step = next(step for step in recipe["steps"] if step["label"] == "python")
    directory = Path(step["environment"]["NEISTRA_FIXTURE_QUALIFICATION_EVIDENCE"])
    assert directory.is_dir() and not directory.is_symlink()
    driver_path = root / "tests/support/ligand_profiles_fixture_qualification.py"
    driver = runpy.run_path(str(driver_path))
    provenance = driver["qualification_provenance"](
        root / "tests/fixtures/ligand-properties-performance/editable"
    )
    reports = {}
    for path in sorted(directory.glob("*/results.json")):
        assert not path.is_symlink()
        body = path.read_bytes()
        result = json.loads(body)
        assert result["complete"] is True and result["count"] == 1
        assert result["provenance"] == provenance
        assert len(result["cases"]) == 1
        case = result["cases"][0]
        name = case["id"]
        assert name == path.parent.name and name not in reports
        assert case["sha256"] == provenance["fixtures"][name]["sha256"]
        history = case["history"]
        assert history["original_preserved"] is True
        assert history["canonical_bytes_ids_graph_coordinates_restored"] is True
        assert history["before_artifact_id"] == history["restored_artifact_id"]
        assert history["edited_artifact_id"] != history["before_artifact_id"]
        assert history["original_sha256"] == case["sha256"]
        store = path.parent / name
        assert store.is_dir() and not store.is_symlink() and store.resolve() == store
        inventory = {}
        artifacts = {}
        for member in sorted(store.rglob("*")):
            assert not member.is_symlink(), "A retained fixture store contains an alias"
            if not member.is_file():
                continue
            payload = member.read_bytes()
            relative = member.relative_to(store).as_posix()
            inventory[relative] = {"bytes": len(payload), "sha256": digest(payload)}
            if relative.startswith("artifacts/sha256/"):
                parts = relative.split("/")
                assert len(parts) == 4 and re.fullmatch(r"[0-9a-f]{64}", parts[3])
                assert parts[2] == parts[3][:2] and inventory[relative]["sha256"] == parts[3]
                artifacts[parts[3]] = relative
        assert "history.db" in inventory, "Missing authentic history database"
        with (store / "history.db").open("rb") as database:
            assert database.read(16) == b"SQLite format 3\0", "Invalid retained database bytes"
        assert len(artifacts) >= 3 and case["sha256"] in artifacts, (
            "Missing original or before/edited normalized artifacts"
        )
        reports[name] = {
            "path": str(path),
            "bytes": len(body),
            "sha256": digest(body),
            "store": str(store),
            "store_files": inventory,
        }
    assert set(reports) == set(provenance["fixtures"]) and len(reports) == 44
    return {"directory": str(directory), "count": 44, "provenance": provenance, "reports": reports}


def validate_benchmark_order(inventory: dict[str, Any]) -> dict[str, Any]:
    """Keep the genuinely cold inherited campaign ahead of the complementary work."""
    assert not inventory.get("errors")
    config = inventory["config"]
    assert type(config["workers"]) is int and config["workers"] == 1
    assert config["fullyParallel"] is False
    assert all(p.get("fullyParallel", False) is False for p in config["projects"])
    records: list[tuple[str, str, str]] = []

    def walk(suite: dict[str, Any]) -> None:
        for spec in suite.get("specs", []):
            for test in spec["tests"]:
                records.append(
                    (test["projectName"], Path(spec["file"]).name, test["expectedStatus"])
                )
        for child in suite.get("suites", []):
            walk(child)

    for suite in inventory["suites"]:
        walk(suite)
    assert records
    result = {}
    inherited = ("ligand-profiles-performance.spec.ts", "ligand-properties-performance.spec.ts")
    complementary = "ligand-sidebar-performance.spec.ts"
    for project in sorted({record[0] for record in records}):
        cases = [(name, status) for owner, name, status in records if owner == project]
        starts = [i for i, (name, _) in enumerate(cases) if name == complementary]
        assert len(starts) == 1 and cases[starts[0]][1] == "passed", project
        ends = {}
        for name in inherited:
            indexes = [i for i, (file, _) in enumerate(cases) if file == name]
            assert indexes and max(indexes) < starts[0], "Complement can prewarm the cold campaign"
            ends[name] = max(indexes)
        result[project] = {"inherited_last_indexes": ends, "complement_first_index": starts[0]}
    return {"workers": 1, "fully_parallel": False, "projects": result}

    assert recipe["compatibility_protocol"] == "private-derived-same-schema-v2"
    assert recipe["preceding_reader"]["tag"] == f"v{PREVIOUS_VERSION}"
    assert recipe["preceding_reader"]["version"] == PREVIOUS_VERSION
    assert recipe["preceding_reader"]["schema_head"] == "0017"
    assert tuple(step["label"] for step in recipe["steps"]) == REQUIRED_STEPS
    builder = runpy.run_path(str(root / TOOL_DIRECTORY / "release-recipe.py.txt"))
    expected = builder["build_recipe"](
        root, recipe["label"], CURRENT_VERSION, recipe["preceding_reader"]
    )
    assert recipe == expected, "Recipe responsibilities, command authority or environment changed"


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("recipe", type=Path)
    args = parser.parse_args()
    root = Path.cwd().resolve()
    recipe = json.loads(args.recipe.read_text())
    validate_recipe_contract(recipe, root)
    baseline = preceding_browser_baseline(root, recipe["preceding_reader"])
    data_before = qualification_data_inventory(root)
    verify_current_versions(root)
    out = root / TOOL_DIRECTORY
    manifest = out / f"{recipe['label']}-gate.json"
    assert not manifest.exists(), "Preserve prior attempts; use a new recipe label."
    initial_status = subprocess.check_output(["git", "status", "--porcelain=v1"], text=True)
    assert recipe["require_clean"] and not initial_status, "Clean committed candidate required."
    preceding = recipe["preceding_reader"]
    assert (
        preceding["commit"]
        == subprocess.check_output(
            ["git", "rev-parse", f"{preceding['tag']}^{{}}"],
            text=True,
        ).strip()
    )
    expected_old = verify_predecessor_catalogue(
        root, preceding, Path(preceding["receipt"]), baseline
    )
    old_root = Path(preceding["root"])
    old_before = extracted_identities(old_root, expected_old)
    assert old_before == expected_old, "Preceding reader does not match the actual tagged source."
    state = {
        "label": recipe["label"],
        "compatibility_protocol": recipe["compatibility_protocol"],
        "commit": subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip(),
        "branch": subprocess.check_output(["git", "branch", "--show-current"], text=True).strip(),
        "initial_status": initial_status,
        "started_utc": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
        "platform": platform.platform(),
        "recipe_sha256": digest(args.recipe.read_bytes()),
        "runner_sha256": digest(Path(__file__).read_bytes()),
        "unset_environment": recipe["unset_environment"],
        "source_before": current_identities(root),
        "preceding_reader": preceding,
        "preceding_source_expected": expected_old,
        "preceding_source_before": old_before,
        "preceding_browser_baseline": baseline,
        "qualification_data_before": data_before,
        "steps": [],
        "complete": False,
        "qualification_profile": "portable correctness; no strict reference-host timing claim",
    }
    manifest.write_text(json.dumps(state, indent=2) + "\n")
    for step in recipe["steps"]:
        print("START", step["label"], flush=True)
        started = time.monotonic()
        inherited = {k: v for k, v in os.environ.items() if k not in recipe["unset_environment"]}
        environment = {**inherited, **recipe["environment"], **step.get("environment", {})}
        if step["label"] == "browser":
            inventory = json.loads((out / f"{recipe['label']}-browser-inventory.json").read_bytes())
            state["benchmark_order"] = validate_benchmark_order(inventory)
            validate_fresh_browser_store(
                Path(environment["NEISTRA_E2E_DATA_DIR"]),
                Path("/tmp") / f"neistra161-{recipe['label']}",
            )
            state["browser_store_initially_absent"] = True
            manifest.write_text(json.dumps(state, indent=2) + "\n")
            # The browser configuration can reuse a server. Qualification must
            # start its own exact-candidate services rather than reuse one.
            for key in ("NEISTRA_E2E_API_PORT", "NEISTRA_E2E_WORKER_PORT", "NEISTRA_E2E_WEB_PORT"):
                with socket.socket() as probe:
                    probe.settimeout(1)
                    assert probe.connect_ex(("127.0.0.1", int(environment[key]))) != 0, (
                        f"Qualification port is already occupied: {key}={environment[key]}"
                    )
        proc = subprocess.Popen(
            step["command"],
            cwd=step.get("cwd", str(root)),
            env=environment,
            stdout=subprocess.PIPE,
            stderr=subprocess.STDOUT,
            start_new_session=True,
        )
        chunks = []
        for line in proc.stdout:
            chunks.append(line)
            clean = re.sub(r"\x1b\[[0-9;]*[A-Za-z]", "", line.decode(errors="replace")).strip()
            if re.match(r"^\[\d+/\d+\]", clean) or clean.startswith(("✓", "✘", "Running ")):
                print(clean, flush=True)
        code = proc.wait()
        content = b"".join(chunks)
        log = out / f"{recipe['label']}-{step['label']}.log.gz"
        log.write_bytes(gzip.compress(content, mtime=0))
        state["steps"].append(
            {
                **step,
                "exit_code": code,
                "isolated_child_pid": proc.pid,
                "isolated_session_requested": True,
                "seconds": round(time.monotonic() - started, 3),
                "log": str(log.relative_to(root)),
                "log_sha256": digest(content),
            }
        )
        state["source_after"] = current_identities(root)
        state["source_unchanged"] = state["source_before"] == state["source_after"]
        state["qualification_data_after"] = qualification_data_inventory(root)
        state["qualification_data_unchanged"] = (
            state["qualification_data_before"] == state["qualification_data_after"]
        )
        state["preceding_source_after"] = extracted_identities(old_root, expected_old)
        state["preceding_source_unchanged"] = state["preceding_source_after"] == expected_old
        manifest.write_text(json.dumps(state, indent=2) + "\n")
        print("PASS" if code == 0 else "FAIL", step["label"], flush=True)
        print(content.decode(errors="replace")[-1200:], flush=True)
        if code:
            raise SystemExit(code)
        assert state["source_unchanged"], "Current source inventory changed during qualification."
        assert state["qualification_data_unchanged"], "Preceding qualification data changed."
        assert state["preceding_source_unchanged"], "Preceding source changed during qualification."
    state["complete"] = True
    state["finished_utc"] = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
    state["final_status"] = subprocess.check_output(["git", "status", "--porcelain=v1"], text=True)
    manifest.write_text(json.dumps(state, indent=2) + "\n")
    print("COMPLETE", manifest, flush=True)


if __name__ == "__main__":
    main()

"""Build a self-verifying public bundle from completed qualification only."""
from pathlib import Path
import hashlib, io, json, sqlite3, subprocess, zipfile
root=Path.cwd();out=root/'docs/assets/compact-modeling-toolbar'
c=json.loads((out/'c3-candidate2-gate.json').read_text());m=json.loads((out/'c4-merged1-gate.json').read_text())
assert c['complete'] and m['complete'] and len(c['steps'])==len(m['steps'])==20
assert c['source_before']==m['source_before']==m['source_after']
sha=subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip();assert sha==m['commit']
s=json.loads((out/'c4-merged1-summary.json').read_text());assert s['complete']
qualification={'issue':93,'runtime_commit':sha,'candidate_commit':c['commit'],'source_identity_count':s['source_count'],'source_sha256':s['source_sha256'],'all_20_commands_pass_both_phases':True,'previous_reader':'actual tagged0.24.0','version':'0.24.1','migration_head':'0014','dependency_upgrade':False,'python':s['python'],'frontend':s['frontend'],'supervisor':s['supervisor'],'browser':s['browser']['expected'],'browser_exclusions':s['browser']['skipped'],'failures':0,'flakes':0,'retries':0,'native_scientific_skips':0}
files={}
with zipfile.ZipFile(io.BytesIO(subprocess.check_output(['git','archive','--format=zip',sha]))) as archive:
 for name in archive.namelist():
  if not name.endswith('/'):files['source/'+name]=archive.read(name)
for name,digest in m['source_before'].items():assert hashlib.sha256(files['source/'+name]).hexdigest()==digest
for p in out.iterdir():
 if p.is_file():files['evidence/'+p.name]=p.read_bytes()
for phase,gate in [('candidate',c),('merged',m)]:
 task=Path(gate['environment']['MOLWEAVE_DATA_DIR']).parent
 for folder in ['forward','reverse','retained-native']:
  for p in (task/folder).rglob('*'):
   if p.is_file() and not p.name.endswith(('-wal','-shm')):
    content=p.read_bytes()
    if p.name=='molweave.db':
     with sqlite3.connect(f'file:{p}?mode=ro',uri=True) as src,sqlite3.connect(':memory:') as dst:
      src.backup(dst);content=dst.serialize()
    files[f'compatibility/{phase}/{folder}/'+str(p.relative_to(task/folder))]=content
files['qualification.json']=(json.dumps(qualification,indent=2)+'\n').encode()
files['release-notes.md']=Path('/tmp/neistra93-release.md').read_bytes()
manifest={name:hashlib.sha256(data).hexdigest() for name,data in sorted(files.items())}
files['SHA256.json']=(json.dumps(manifest,indent=2)+'\n').encode()
target=Path('/datadisk/dev/.neistra-tools/neistra-v0.24.1-verification.zip');assert not target.exists()
with zipfile.ZipFile(target,'w',zipfile.ZIP_DEFLATED) as archive:
 for name,data in sorted(files.items()):archive.writestr(name,data)
with zipfile.ZipFile(target) as archive:
 assert archive.testzip() is None
 assert all(hashlib.sha256(archive.read(n)).hexdigest()==v for n,v in manifest.items())
result={'filename':target.name,'path':str(target),'size':target.stat().st_size,'sha256':hashlib.sha256(target.read_bytes()).hexdigest(),'members':len(files),'internal_hashes':len(manifest),'crc_verified':True,'qualification':qualification}
Path('/tmp/neistra93-bundle.json').write_text(json.dumps(result,indent=2)+'\n');print(json.dumps(result,indent=2))

def walk(suite, output, parents=(), *, inventory=False):
    for spec in suite.get("specs", []):
        for test in spec["tests"]:
            result = None if inventory else test["results"][-1]
            title = " \u203a ".join((*parents, spec["title"]))
            output.append(
                {
                    "identity": (
                        f"[{test['projectName']}] \u203a tests/e2e/{spec['file']} \u203a {title}"
                    ),
                    "file": spec["file"],
                    "project": test["projectName"],
                    "title": title,
                    "status": None if result is None else result["status"],
                    "expected_status": test["expectedStatus"],
                    "retries": 0 if inventory else len(test["results"]) - 1,
                }
            )
    for child in suite.get("suites", []):
        walk(
            child,
            output,
            parents + ((suite["title"],) if suite.get("specs") else ()),
            inventory=inventory,
        )

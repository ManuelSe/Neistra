"""Issue 95 command evidence: exact argv/environment, source identities and lossless logs."""
import argparse
import gzip
import hashlib
import json
import os
from pathlib import Path
import platform
import subprocess
import time

parser = argparse.ArgumentParser()
parser.add_argument("label")
parser.add_argument("commands", type=Path)
args = parser.parse_args()
root = Path.cwd()
output = root / "docs/assets/workspace-sequence-viewer"
output.mkdir(parents=True, exist_ok=True)
manifest = output / f"{args.label}-gate.json"
assert not manifest.exists(), "Use a new evidence label; previous attempts are retained."
recipe = json.loads(args.commands.read_text())
files = subprocess.check_output(["git", "ls-files", "--cached", "--others", "--exclude-standard"], text=True).splitlines()
files = sorted({p for p in files if p.startswith(("apps/", "packages/", "tests/", "tools/", "patches/")) or p in {"pyproject.toml", "uv.lock", "package.json", "pnpm-lock.yaml", "playwright.config.ts"}})
def identities():
    return {p: hashlib.sha256((root / p).read_bytes()).hexdigest() for p in files if (root / p).is_file()}
environment = {**os.environ, **recipe.get("environment", {})}
metadata = {
    "label": args.label,
    "commit": subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip(),
    "branch": subprocess.check_output(["git", "branch", "--show-current"], text=True).strip(),
    "initial_status": subprocess.check_output(["git", "status", "--porcelain=v1"], text=True),
    "started_at": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
    "platform": platform.platform(),
    "qualifier_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
    "environment": {k: environment[k] for k in recipe.get("environment", {})},
    "source_before": identities(),
    "steps": [],
    "complete": False,
}
manifest.write_text(json.dumps(metadata, indent=2) + "\n")
for step in recipe["steps"]:
    print("START", step["label"], flush=True)
    started = time.monotonic()
    completed = subprocess.run(step["command"], env={**environment, **step.get("environment", {})}, stdout=subprocess.PIPE, stderr=subprocess.STDOUT)
    content = completed.stdout
    log = output / f'{args.label}-{step["label"]}.log.gz'
    log.write_bytes(gzip.compress(content, mtime=0))
    record = {**step, "exit_code": completed.returncode, "seconds": round(time.monotonic()-started, 3), "log": str(log.relative_to(root)), "log_sha256": hashlib.sha256(content).hexdigest()}
    metadata["steps"].append(record)
    metadata["source_after"] = identities()
    metadata["source_unchanged"] = metadata["source_before"] == metadata["source_after"]
    manifest.write_text(json.dumps(metadata, indent=2) + "\n")
    print("PASS" if completed.returncode == 0 else "FAIL", step["label"], record["seconds"], flush=True)
    print(content.decode(errors="replace")[-2000:], flush=True)
    if completed.returncode != 0:
        raise SystemExit(completed.returncode)
    assert metadata["source_unchanged"], "Source changed while checks were running."
metadata["complete"] = True
metadata["finished_at"] = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
manifest.write_text(json.dumps(metadata, indent=2) + "\n")
print("COMPLETE", manifest.relative_to(root), flush=True)

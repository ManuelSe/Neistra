"""Verify actual issue95 publication receipts; never manufacture remote state."""
from pathlib import Path
import argparse, hashlib, json, subprocess, zipfile

p=argparse.ArgumentParser()
for name in ['merged','runtime-pr','comment','notes','bundle','download','download-receipt','closeout','review']:
 p.add_argument('--'+name,required=True)
a=p.parse_args()
out=Path('docs/assets/workspace-sequence-viewer');repo='repos/ManuelSe/Neistra'
def gh(*args):return json.loads(subprocess.check_output(['gh',*args],text=True))
g=json.loads((out/f'{a.merged}-gate.json').read_text());sha=g['commit']
assert g['complete'] and g['source_unchanged'] and len(g['steps'])==20
assert subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip()==sha
pr=gh('pr','view',a.runtime_pr,'--json','number,url,state,mergedAt,mergeCommit,headRefOid')
assert pr['state']=='MERGED' and pr['mergeCommit']['oid']==sha
release=gh('api',repo+'/releases/tags/v0.26.0')
assert not release['draft'] and not release['prerelease'] and release['target_commitish']==sha
assert gh('api',repo+'/releases/latest')['id']==release['id']
assert release['body'].strip()==Path(a.notes).read_text().strip()
ref=gh('api',repo+'/git/ref/tags/v0.26.0');assert ref['object']['type']=='tag'
tag=gh('api',repo+'/git/tags/'+ref['object']['sha'])
assert tag['object']['type']=='commit' and tag['object']['sha']==sha
assert subprocess.check_output(['git','ls-remote','origin','refs/tags/v0.26.0^{}'],text=True).split()[0]==sha
bundle=json.loads(Path(a.bundle).read_text())
assert bundle['qualification']['issue']==95 and bundle['qualification']['runtime_commit']==sha
assert bundle['qualification']['version']=='0.26.0' and bundle['qualification']['merged_label']==a.merged
asset=next(v for v in release['assets'] if v['name']==bundle['filename'])
assert asset['digest']=='sha256:'+bundle['sha256']
receipt=json.loads(Path(a.download_receipt).read_text())
assert receipt['public_download_verified'] and receipt['url']==asset['browser_download_url']
assert receipt['asset_id']==asset['id'] and receipt['sha256']==bundle['sha256']
assert hashlib.sha256(Path(a.download).read_bytes()).hexdigest()==bundle['sha256']
with zipfile.ZipFile(a.download) as z:
 assert z.testzip() is None
 manifest=json.loads(z.read('SHA256.json'));assert len(manifest)==bundle['internal_hashes']
 assert all(hashlib.sha256(z.read(n)).hexdigest()==v for n,v in manifest.items())
 assert len(z.namelist())==bundle['members']
bundle.update(public_url=asset['browser_download_url'],asset_id=asset['id'],remote_digest=asset['digest'],
 public_published_download_verified=True,downloaded_zip_crc_and_all_internal_hashes_verified=True)
comment=gh('api',repo+'/issues/comments/'+a.comment)
assert comment['body'].strip()==Path(a.closeout).read_text().strip()
assert gh('issue','view','95','--json','state')['state']=='CLOSED'
review=json.loads(Path(a.review).read_text())
assert review['unresolved_review_threads']==0 and not review['override_used']
record={'status':'runtime release and issue closeout verified; final docs/ref audit follows at linked issue reply',
 'issue':95,'plan':'docs/plans/issue-95-workspace-sequence-viewer.md',
 'runtime_pr':{**pr,'strategy':'normal merge preserving checkpoint history'},
 'tag':{'name':'v0.26.0','type':'annotated','object':ref['object']['sha'],'commit':sha,
 'remote_ref_and_peeled_commit_verified':True},
 'release':{k:release[k] for k in ['id','html_url','tag_name','target_commitish','draft','prerelease','published_at']},
 'release_notes_sha256':hashlib.sha256(release['body'].encode()).hexdigest(),
 'verification_bundle':bundle,'issue_closeout':{'id':comment['id'],'url':comment['html_url'],
 'verified_body_sha256':hashlib.sha256(comment['body'].encode()).hexdigest(),
 'final_reference_audit':'Final documentation integration and actual cleanup are appended after verification.'},
 'qualification':bundle['qualification'],'review':review,
 'compatibility':{'database_head':'0014','normalized_selection_component_schema':1,
 'project_viewer_archive_schema':2,'new_migration':False,'dependency_version_upgrade':False},
 'new_followup_issues':[],'feature_branch':'feat/issue-95-workspace-sequence-viewer',
 'cleanup_order':'Delete only after verified runtime merge, tag, release/public bundle, issue reply and docs audit. Runtime tag remains fixed.'}
(out/'c4-publication.json').write_text(json.dumps(record,indent=2)+'\n')
print(json.dumps({'runtime':sha,'pr':pr['url'],'release':release['html_url'],
 'issue_reply':comment['html_url'],'bundle_sha256':bundle['sha256']},indent=2))

"""Audit five application versions, unchanged dependencies and durable schema identities."""

import json
import re
import subprocess
import tomllib
from pathlib import Path

from alembic.config import Config
from alembic.script import ScriptDirectory
from neistra_api.archive_service import ARCHIVE_SCHEMA_VERSION, ProjectManifestV2
from neistra_api.schemas import ProjectRead, ViewerSettingsV2
from neistra_core.components import ComponentHierarchyV1
from neistra_core.minimization.quick_plugin import GRAPH_POLICY, IMPLEMENTATION_VERSION
from neistra_core.molecular import NormalizedStructureV1
from neistra_core.selection import SelectionV1

BASE = "a1f2e2fd92d19e4a85b3237b41f2d552eff3a0d9"
VERSION = "0.30.0"
project = tomllib.loads(Path("pyproject.toml").read_text())["project"]
lock = tomllib.loads(Path("uv.lock").read_text())
root_package = next(p for p in lock["package"] if p["name"] == "neistra-dev")
api = re.search(r'version="([^"]+)"', Path("apps/api/src/neistra_api/main.py").read_text())[1]
archive = re.search(
    r'APPLICATION_VERSION = "([^"]+)"',
    Path("apps/api/src/neistra_api/archive_service.py").read_text(),
)[1]
web = json.loads(Path("apps/web/package.json").read_text())["version"]
assert {project["version"], root_package["version"], api, archive, web} == {VERSION}
assert project["name"] == "neistra-dev"
baseline = tomllib.loads(subprocess.check_output(["git", "show", f"{BASE}:uv.lock"], text=True))
assert {k: v for k, v in lock.items() if k != "package"} == {
    k: v for k, v in baseline.items() if k != "package"
}
assert [p for p in lock["package"] if p["name"] != "neistra-dev"] == [
    p for p in baseline["package"] if p["name"] != "neistra-dev"
]
assert {k: v for k, v in root_package.items() if k != "version"} == {
    k: v
    for k, v in next(p for p in baseline["package"] if p["name"] == "neistra-dev").items()
    if k != "version"
}
base_project = tomllib.loads(
    subprocess.check_output(
        ["git", "show", f"{BASE}:pyproject.toml"],
        text=True,
    )
)
current_project = tomllib.loads(Path("pyproject.toml").read_text())
base_project["project"].pop("version")
current_project["project"].pop("version")
assert base_project == current_project
assert not subprocess.check_output(
    [
        "git",
        "diff",
        BASE,
        "--",
        "pnpm-lock.yaml",
        "pnpm-workspace.yaml",
        "package.json",
        "tools/ambertools-linux-64.lock",
    ],
    text=True,
)
migrations = Path("apps/api/migrations/versions")
historical = subprocess.check_output(
    ["git", "ls-tree", "-r", "--name-only", BASE, "--", str(migrations)], text=True
).splitlines()
assert len(historical) == 14
for name in historical:
    original = subprocess.check_output(["git", "show", f"{BASE}:{name}"])
    assert Path(name).read_bytes() == original, name
assert {str(p) for p in migrations.glob("*.py")} - set(historical) == {
    "apps/api/migrations/versions/0015_workspace_presentation.py"
}
base_web = json.loads(
    subprocess.check_output(
        ["git", "show", f"{BASE}:apps/web/package.json"],
        text=True,
    )
)
current_web = json.loads(Path("apps/web/package.json").read_text())
assert {k: v for k, v in base_web.items() if k != "version"} == {
    k: v for k, v in current_web.items() if k != "version"
}
assert ARCHIVE_SCHEMA_VERSION == 2
for cls, version in (
    (ProjectManifestV2, 2),
    (ProjectRead, 2),
    (ViewerSettingsV2, 2),
    (NormalizedStructureV1, 1),
    (SelectionV1, 1),
    (ComponentHierarchyV1, 1),
):
    assert cls.model_fields["schema_version"].default == version, cls
scripts = ScriptDirectory.from_config(Config("alembic.ini"))
assert scripts.get_heads() == ["0015"]
assert scripts.get_revision("0015").down_revision == "0014"
assert IMPLEMENTATION_VERSION == "1.0.3" and GRAPH_POLICY == "current-canonical-v1"
print(
    json.dumps(
        {
            "version": VERSION,
            "five_sources_agree": True,
            "dependencies_unchanged": True,
            "historical_migrations_unchanged": True,
            "archive_project_viewer": 2,
            "normalized_selection_component": 1,
            "new_migration": "0015_workspace_presentation.py",
            "head": "0015",
            "quick_implementation": IMPLEMENTATION_VERSION,
            "quick_graph_policy": GRAPH_POLICY,
            "comparison_base": BASE,
        },
        indent=2,
    )
)

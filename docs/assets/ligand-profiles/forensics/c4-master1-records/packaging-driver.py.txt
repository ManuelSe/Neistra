def copy_exact(source, destination):
    assert source.is_file() and not source.is_symlink(), source
    body = source.read_bytes()
    if destination.exists():
        assert destination.is_file() and not destination.is_symlink()
        assert destination.read_bytes() == body, destination
    else:
        destination.parent.mkdir(parents=True, exist_ok=True)
        destination.write_bytes(body)
    return {'bytes': len(body), 'sha256': sha(body)}

def git_bodies(commit, names):
    output = subprocess.check_output(['git', 'cat-file', '--batch'], cwd=FAILED,
        input=''.join(f'{commit}:{name}\n' for name in names).encode())
    result = {}
    offset = 0
    for name in names:
        end = output.index(b'\n', offset)
        _, kind, size = output[offset:end].split()
        assert kind == b'blob', name
        start = end + 1
        offset = start + int(size)
        result[name] = output[start:offset]
        assert output[offset:offset+1] == b'\n'
        offset += 1
    assert offset == len(output)
    return result

def tree(prefix, base):
    assert base.is_dir() and not base.is_symlink()
    names = set()
    empty_directories[prefix] = []
    for path in sorted(base.rglob('*')):
        assert not path.is_symlink(), path
        relative = path.relative_to(base).as_posix()
        if path.is_file():
            name = prefix + '/' + relative
            add(name, path)
            names.add(name)
        elif path.is_dir() and not any(path.iterdir()):
            empty_directories[prefix].append(relative)
    trees.append((prefix, base, names))
    return len(names)

scanned = 0
def scan_body(name, body, depth=0):
    global scanned
    assert depth < 16
    assert not any(token in name.lower().encode() + b'\n' + body.lower() for token in forbidden), ('restricted reference', name)
    scanned += 1
    if body.startswith(b'\x1f\x8b'):
        scan_body(name + '!gunzip', gzip.decompress(body), depth+1)
    elif zipfile.is_zipfile(io.BytesIO(body)):
        with zipfile.ZipFile(io.BytesIO(body)) as archive:
            assert archive.testzip() is None
            assert len(archive.namelist()) == len(set(archive.namelist()))
            for member in archive.infolist():
                if not member.is_dir(): scan_body(name + '!' + member.filename, archive.read(member), depth+1)

def partition_assets(rows,directory):
    names = ('gate-parts','retained-state','case-companions')
    grouped = {name:[] for name in names}
    parents = {directory/name:name for name in names}
    seen_names,seen_paths = set(),set()
    for row in rows:
        path = Path(row['path'])
        assert path.is_absolute() and path == path.resolve() and not path.is_symlink(), 'Noncanonical or symlink part path'
        assert path.parent in parents, 'Orphan asset outside the three complete evidence groups'
        assert path.name == row['asset_name'] and row['asset_name'] not in seen_names and path not in seen_paths, 'Duplicate or mismatched asset identity'
        seen_names.add(row['asset_name'])
        seen_paths.add(path)
        grouped[parents[path.parent]].append(row)
    assert all(grouped.values()), 'Every complete evidence group needs at least one part'
    return grouped

def fp(body):
    if isinstance(body,Path):
        digest = hashlib.sha256()
        size = 0
        with body.open('rb') as handle:
            while block := handle.read(1024*1024):
                size += len(block)
                digest.update(block)
        return {'bytes':size,'sha256':digest.hexdigest()}
    return {'bytes':len(body),'sha256':hashlib.sha256(body).hexdigest()}

"""Audit docs-only publication changes without rerunning identical runtime."""
from pathlib import Path
import datetime,gzip,hashlib,json,re,subprocess
out=Path('docs/assets/protein-residue-hierarchy')
c=json.loads((out/'c3-candidate3-gate.json').read_text());m=json.loads((out/'c4-merged-gate.json').read_text())
assert c['complete'] and m['complete'] and c['source_before']==m['source_before']==m['source_after']
source=m['source_before'];assert len(source)==571
assert all(hashlib.sha256(Path(n).read_bytes()).hexdigest()==v for n,v in source.items())
changes=subprocess.check_output(['git','diff','--name-only',m['commit']],text=True).splitlines();assert changes and all(n.startswith('docs/') for n in changes)
assert not subprocess.check_output(['git','diff','--name-only',m['commit'],'--','docs/PLAN.md'],text=True)
links=[]
for name in changes:
 if not name.endswith('.md'):continue
 diff=subprocess.check_output(['git','diff','--unified=0',m['commit'],'--',name],text=True)
 for line in diff.splitlines():
  if not line.startswith('+') or line.startswith('+++'):continue
  for ref in re.findall(r'\]\(([^)]+)\)',line):
   if '://' in ref or ref.startswith('#'):continue
   assert (Path(name).parent/ref.split('#')[0]).resolve().exists(),(name,ref)
   links.append((name,ref))
jcount=gcount=0
for p in out.glob('*.json'):json.loads(p.read_text());jcount+=1
for p in out.glob('*.gz'):gzip.decompress(p.read_bytes());gcount+=1
for label in ['c3-candidate3','c4-merged']:
 gate=json.loads((out/f'{label}-gate.json').read_text());assert len(gate['steps'])==20 and gate['complete'] and gate['source_unchanged']
 for step in gate['steps']:
  assert step['exit_code']==0
  assert hashlib.sha256(gzip.decompress(Path(step['log']).read_bytes())).hexdigest()==step['log_sha256']
 summary=json.loads((out/f'{label}-summary.json').read_text());assert not summary['retries'] and not summary['browser']['unexpected'] and not summary['browser']['flaky']
subprocess.run(['.venv/bin/python','docs/assets/protein-residue-hierarchy/versions.py.txt'],check=True)
subprocess.run(['git','diff','--check'],check=True)
result={'checked_at':datetime.datetime.now(datetime.UTC).isoformat(),'script':'docs/assets/protein-residue-hierarchy/documentation-check.py.txt','script_sha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),'runtime':m['commit'],'runtime_test_patch_tool_lock_identities_match':571,'source_digest':hashlib.sha256(json.dumps(source,sort_keys=True).encode()).hexdigest(),'changed_tracked_files_only_docs':changes,'global_plan_unchanged':True,'new_relative_links_checked':links,'asset_json_parsed':jcount,'gzip_crc_checked':gcount,'both_complete_gate_log_hashes_verified':True,'versions_dependencies_production_harness_audit_passed':True,'diff_check_passed':True,'new_runtime_tests_required':False,'rationale':'Only documentation/evidence changed after the complete exact-runtime gate; all571 source/test/tool/patch/lock identities match.'}
(out/'c4-documentation-check.json').write_text(json.dumps(result,indent=2)+'\n');print(json.dumps({'passed':True,'links':len(links),'json':jcount,'gzip':gcount}))
